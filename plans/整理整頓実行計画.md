```go
// back/internal/presentation/http/router/router.go
package router

import (
    "github.com/gin-gonic/gin"
    "github.com/gin-contrib/cors"
)

func NewRouter(
    authHandler *handler.AuthHandler,
    problemHandler *handler.ProblemHandler,
    sseHandler *handler.SSEHandler,
) *gin.Engine {
    r := gin.Default()
    
    // CORS設定
    r.Use(cors.New(cors.Config{
        AllowOrigins:     []string{"http://localhost:3000"},
        AllowMethods:     []string{"GET", "POST", "PUT", "DELETE", "OPTIONS"},
        AllowHeaders:     []string{"Origin", "Content-Type", "Authorization"},
        AllowCredentials: true,
    }))
    
    // ヘルスチェック
    r.GET("/health", func(c *gin.Context) {
        c.JSON(200, gin.H{"status": "ok"})
    })
    
    // 公開API
    public := r.Group("/api")
    {
        public.POST("/login", authHandler.Login)
        public.POST("/forgot-password", authHandler.ForgotPassword)
    }
    
    // 認証必須API
    authorized := r.Group("/api")
    authorized.Use(middleware.RequireAuth())
    {
        // 問題生成
        authorized.POST("/generate-problem", problemHandler.GenerateProblem)
        authorized.POST("/generate-problem-five-stage-sse", sseHandler.GenerateFiveStageSSE)
        authorized.POST("/generate-three-problems-sse", sseHandler.GenerateThreeProblemsSSE)
        
        // 問題管理
        problems := authorized.Group("/problems")
        {
            problems.GET("/search", problemHandler.SearchProblems)
            problems.POST("/search-by-filters", problemHandler.SearchByFilters)
            problems.POST("/search-combined", problemHandler.SearchCombined)
            problems.GET("/history", problemHandler.GetHistory)
            problems.PUT("/update", problemHandler.Update)
            problems.POST("/regenerate-geometry", problemHandler.RegenerateGeometry)
        }
        
        // ユーザー管理
        user := authorized.Group("/user")
        {
            user.GET("/info", authHandler.GetUserInfo)
            user.GET("/profile", authHandler.GetProfile)
            user.PUT("/settings", authHandler.UpdateSettings)
        }
    }
    
    return r
}
```

```go
// back/internal/presentation/http/handler/problem_handler.go
package handler

import (
    "net/http"
    "github.com/gin-gonic/gin"
)

type ProblemHandler struct {
    generateUseCase *problem.GenerateUseCase
    searchUseCase   *problem.SearchUseCase
    logger          logger.Logger
}

func (h *ProblemHandler) GenerateProblem(c *gin.Context) {
    var req problem.GenerateRequest
    if err := c.ShouldBindJSON(&req); err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
        return
    }
    
    result, err := h.generateUseCase.Execute(c.Request.Context(), req)
    if err != nil {
        h.logger.Error("Failed to generate problem", zap.Error(err))
        c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
        return
    }
    
    c.JSON(http.StatusOK, result)
}
```

```go
package user

type User struct {

    role                    Role
    preferredAI             AIProvider

    problemGenerationCount  Count
    createdAt               time.Time

    if u.problemGenerationLimit.IsUnlimited() {
        return true
    }
    return u.problemGenerationCount.Value() < u.problemGenerationLimit.Value()
}

func (u *User) IncrementGenerationCount() error {
    if !u.CanGenerateProblem() {
        return errors.New("問題生成回数の上限に達しました")

    u.problemGenerationCount = u.problemGenerationCount.Increment()
    return nil
}
```

#### Day 1-3: GenerateUseCase

```go
// back/internal/application/problem/generate_usecase.go
package problem

type GenerateUseCase struct {
    strategyFactory generation.StrategyFactory
    problemRepo     domain.ProblemRepository
    userRepo        domain.UserRepository
    logger          logger.Logger
}

func (uc *GenerateUseCase) Execute(
    ctx context.Context,
    req GenerateRequest,
) (*GenerateResponse, error) {
    // 1. ユーザー取得
    user, err := uc.userRepo.FindByID(ctx, req.UserID)
    if err != nil {
        return nil, err
    }
    
    // 2. バリデーション
    if !user.CanGenerateProblem() {
        return nil, errors.New("生成回数の上限に達しました")
    }
    
    // 3. 戦略選択
    strategy := uc.strategyFactory.Create(req.Mode)
    
    // 4. 生成実行
    result, err := strategy.Generate(ctx, req.ToGenerationRequest())
    if err != nil {
        return nil, err
    }
    
    // 5. 保存
    if err := uc.problemRepo.Save(ctx, result.Problem); err != nil {
        return nil, err
    }
    
    // 6. カウント更新
    if err := user.IncrementGenerationCount(); err != nil {
        return nil, err
    }
    
    return &GenerateResponse{
        ProblemID: result.Problem.ID(),
        Content:   result.Problem.Content(),
        Solution:  result.Problem.Solution(),
    }, nil
}
```
