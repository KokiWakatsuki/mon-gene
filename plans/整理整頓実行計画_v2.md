package handler

import (
    "net/http"

type ProblemHandler struct {
    generateUC *usecase.GenerateProblem
    searchUC   *usecase.SearchProblem
    logger     shared.Logger
}

func NewProblemHandler(
    generateUC *usecase.GenerateProblem,
    searchUC *usecase.SearchProblem,
    logger shared.Logger,
) *ProblemHandler {
    return &ProblemHandler{
        generateUC: generateUC,
        searchUC:   searchUC,
        logger:     logger,
    }
}

func (h *ProblemHandler) Generate(c *gin.Context) {
    var req GenerateRequest
    if err := c.ShouldBindJSON(&req); err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
        return
    }
    
    result, err := h.generateUC.Execute(c.Request.Context(), req)
    if err != nil {
        h.logger.Error("Failed to generate", zap.Error(err))
        c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
        return
    }
    
    c.JSON(http.StatusOK, result)
}

```go
// back/internal/usecase/generate_problem.go（100行）
package usecase

type GenerateProblem struct {
    strategies  map[string]generation.Strategy
    problemRepo domain.ProblemRepository
    userRepo    domain.UserRepository
    logger      shared.Logger
}

func (uc *GenerateProblem) Execute(ctx context.Context, req Request) (*Response, error) {
    // 1. ユーザー取得
    user, err := uc.userRepo.FindByID(ctx, req.UserID)
    if err != nil {
        return nil, err
    }
    
    // 2. バリデーション
    if !user.CanGenerateProblem() {
        return nil, errors.New("生成回数の上限に達しました")
    }
    
    // 3. 戦略選択
    strategy := uc.strategies[req.Mode]
    
    // 4. 生成実行
    result, err := strategy.Generate(ctx, req.ToGenerationRequest())
    if err != nil {
        return nil, err
    }
    
    // 5. 保存
    if err := uc.problemRepo.Save(ctx, result.Problem); err != nil {
        return nil, err
    }
    
    // 6. カウント更新
    user.IncrementGenerationCount()
    uc.userRepo.Update(ctx, user)
    
    return &Response{ProblemID: result.Problem.ID()}, nil
}
```
